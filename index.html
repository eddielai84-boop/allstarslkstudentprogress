📋 配套 Google Apps Script 代码（粘贴到 Google Sheets 扩展程序 → Apps Script）： 
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var headers = ['评论ID', '记录时间点', '学号', '学生姓名', '年级', '班级', '科目代码', '科目名称', '分数', '课堂专注', '功课完成', '课堂互动', '评估老师', '老师评论内容', '状态/备注'];
    var logTab = ss.getSheetByName('All_Comments_Log') || ss.insertSheet('All_Comments_Log');
    if (logTab.getLastRow() === 0) {
      logTab.appendRow(headers);
      logTab.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#b91c1c').setFontColor('#ffffff');
      logTab.setFrozenRows(1);
    }
    var payload = JSON.parse(e.postData.contents);
    var rows = Array.isArray(payload.records) ? payload.records : [payload];
    rows.forEach(function(r) {
      if (!r || !r.comment) return;
      var cid = r.id || ('CMT-' + r.studentId + '-' + r.subjectKey + '-' + new Date().getTime());
      var lastRow = logTab.getLastRow();
      var existingRow = -1;
      if (lastRow >= 2) {
        var ids = logTab.getRange(2, 1, lastRow - 1, 1).getValues();
        for (var i = 0; i < ids.length; i++) {
          if (String(ids[i][0]) === String(cid)) { existingRow = i + 2; break; }
        }
      }
      if (r.action === 'MASTER_ADMIN_DELETE') {
        if (existingRow !== -1) logTab.deleteRow(existingRow);
        return;
      }
      var vals = [cid, r.timestamp, r.studentId, r.studentName, r.grade, r.className, r.subjectKey, r.subjectName, r.score, r.focus, r.homework, r.participation, r.teacher, r.comment, r.action === 'MASTER_ADMIN_EDIT' ? 'Admin已修改' : '正常同步'];
      if (existingRow !== -1) logTab.getRange(existingRow, 1, 1, vals.length).setValues([vals]);
      else logTab.appendRow(vals);
    });
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: rows.length })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
